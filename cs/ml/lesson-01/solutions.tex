\begin{solution}{ex:ml-1}
  Let's compute $Z=XW+\mathbf{1}_4 \mathbf{b}^{\!\top}$:
  \[
    \begin{gathered}
      X = \begin{pmatrix}1&0&2\\0&3&1\\-1&1&1\\2&1&0\end{pmatrix},\quad
      W = \begin{pmatrix}1&-1\\0&2\\1&1\end{pmatrix}\\
      XW = \begin{pmatrix}3&1\\1&7\\0&4\\2&0\end{pmatrix}\\
      \mathbf{1}_4 \mathbf{b}^{\!\top} = \begin{pmatrix}0.5&-0.5\\0.5&-0.5\\0.5&-0.5\\0.5&-0.5\end{pmatrix}\\
      XW + \mathbf{1}_4 \mathbf{b}^{\!\top} = \begin{pmatrix}3.5&0.5\\1.5&6.5\\0.5&3.5\\1.5&-0.5\end{pmatrix}\\
      Z = \begin{pmatrix}3.5&0.5\\1.5&6.5\\0.5&3.5\\1.5&-0.5\end{pmatrix}\\
      R=\operatorname{relu}(Z)=\begin{pmatrix}3.5&0.5\\1.5&6.5\\0.5&3.5\\1.5&0\end{pmatrix}\\
    \end{gathered}
  \]
  Let's check that the first row of Z equals $\mathbf{x}^{\!\top}_1W+\mathbf{b}^{\!\top}$:
  \[
    \begin{gathered}
      \mathbf{x}^{\!\top}_1=\begin{pmatrix}1&0&2\end{pmatrix}\\
      W = \begin{pmatrix}1&-1\\0&2\\1&1\end{pmatrix}\\
      \mathbf{x}^{\!\top}_1W=\begin{pmatrix}3&1\end{pmatrix}
      \mathbf{b}^{\!\top} = \begin{pmatrix}0.5&-0.5\end{pmatrix}
      \mathbf{x}^{\!\top}_1W+\mathbf{b}^{\!\top}=\begin{pmatrix}3.5&0.5\end{pmatrix}
    \end{gathered}
  \]
  We can see that it is equal to the first row of Z.\\
  (b)
  The shape of Z and R are both $R^{m\times2}$, and $\mathbf{1}_4 \mathbf{b}^{\!\top}$ has the same shape.\\
  (c)
  The number of parameters for this layer is $n\cdot 2+1\cdot 2 = 8$. If $n=784$ and $p=128$, the number of parameters would be $n\cdot p + p = 784\cdot 128 + 128 = 100480$ \\
  (d)
  $R$ has only one entry which is zero, namely the bottom right entry. The map $X \mapsto R$ is not linear in X because of the non-linear activation function. The easiest example of $X$ and $a$ for which $\operatorname{relu}(aX) \ne a\operatorname{relu}(X)$ is $X=\begin{pmatrix}1\end{pmatrix}$ and $a=-1$. The inequality is trivial. The negative entry in $Z$ becomes the zero entry in $R$.
\end{solution}
\begin{solution}{ex:ml-2}
  (a)
  Let's calculate $Z_1$ by $Z_1=XW_1+\mathbf{1}_2\mathbf{b}_1^{\!\top}$:
  \[
    \begin{gathered}
      Z_1=XW_1+\mathbf{1}_2\mathbf{b}_1^{\!\top}\\
      Z_1=\begin{pmatrix}1&2\\3&1\end{pmatrix}\begin{pmatrix}0&1\\1&-1\end{pmatrix}+\begin{pmatrix}1\\1\end{pmatrix}\begin{pmatrix}1&0\end{pmatrix}\\
      Z_1=\begin{pmatrix}2&-1\\1&2\end{pmatrix}+\begin{pmatrix}1\\1\end{pmatrix}\begin{pmatrix}1&0\end{pmatrix}\\
      Z_1=\begin{pmatrix}2&-1\\1&2\end{pmatrix}+\begin{pmatrix}1&0\\1&0\end{pmatrix}\\
      Z_1=\begin{pmatrix}3&-1\\2&2\end{pmatrix}
    \end{gathered}
  \]
  Then let's calculate $R$ by $R=\operatorname{relu}(Z_1)$
  \[
    \begin{gathered}
      R=\operatorname{relu}(Z_1)\\
      R=\begin{pmatrix}3&0\\2&2\end{pmatrix}
    \end{gathered}
  \]
  Then let's calculate $Z_2$ by $Z_2=RW_2+\mathbf{1}_2b_2$:
  \[
    \begin{gathered}
      Z_2=RW_2+\mathbf{1}_2b_2\\
      Z_2=\begin{pmatrix}3&0\\2&2\end{pmatrix}\begin{pmatrix}1\\2\end{pmatrix}+\begin{pmatrix}1\\1\end{pmatrix}0\\
      Z_2=\begin{pmatrix}3&0\\2&2\end{pmatrix}\begin{pmatrix}1\\2\end{pmatrix}\\
      Z_2=\begin{pmatrix}3\\6\end{pmatrix}\\
    \end{gathered}
  \]
  Lastly, let's calculate $L$ by $L=\frac{1}{2m}\norm{Z_2-Y}_F^2$:
  \[
    \begin{gathered}
      m=2\\
      L=\frac{1}{4}\norm{Z_2-Y}_F^2\\
      L=\frac{1}{4}\norm{\begin{pmatrix}1\\5\end{pmatrix}}_F^2\\
      L=\frac{1}{4}\cdot (1^2 + 5^2)\\
      L=\frac{1}{4}\cdot 26\\
      L=6.5\\
    \end{gathered}
  \]

  (b) Now, to compute $\bar Z_2, \bar W_2, \bar b_2, \bar R, \bar Z_1, \bar W_1, \bar{\mathbf{b}_1}$, let's start with $\bar Z_2$:

  \[
    \begin{gathered}
      \bar Z_2=\frac{1}{m}(Z_2-Y)\\
      \bar Z_2=\frac{1}{2}(Z_2-Y)\\
      \bar Z_2=\frac{1}{2}\begin{pmatrix}1\\5\end{pmatrix}\\
      \bar Z_2=\begin{pmatrix}0.5\\2.5\end{pmatrix}\\
    \end{gathered}
  \]
  Using the chain rule, we can define the other adjoints:
  \[
    \begin{gathered}
      \bar W_2=R^{\!\top}\bar Z_2\\
      \bar b_2=\bar Z_2^{\!\top}\mathbf{1}_2\\
      \bar R = \bar Z_2 W_2^{\!\top}\\
      \bar Z_1 = \bar R \odot \mathbf{1}[Z_1>0]\\
      \bar W_1 = X^{\!\top} \bar Z_1\\
      \bar{\mathbf{b}_1}=\bar Z_1^{\!\top}\mathbf{1}_2 
    \end{gathered} 
  \]
  Let's calculate them one by one:
  \[
    \begin{gathered}
      \bar W_2 = \begin{pmatrix}3&0\\2&2\end{pmatrix}^{\!\top} \begin{pmatrix}0.5\\2.5\end{pmatrix}\\
      \bar W_2 = \begin{pmatrix}3&2\\0&2\end{pmatrix} \begin{pmatrix}0.5\\2.5\end{pmatrix}\\
      \bar W_2 = \begin{pmatrix}6.5\\5\end{pmatrix}\\
      \bar b_2=\begin{pmatrix}0.5\\2.5\end{pmatrix}^{\!\top}\begin{pmatrix}1\\1\end{pmatrix} \\
      \bar b_2= \begin{pmatrix}0.5&2.5\end{pmatrix}\begin{pmatrix}1\\1\end{pmatrix}\\
      \bar b_2=3\\
      \bar R =  \begin{pmatrix}0.5\\2.5\end{pmatrix}\begin{pmatrix}1\\2\end{pmatrix}^{\!\top}\\
      \bar R = \begin{pmatrix}0.5\\2.5\end{pmatrix}\begin{pmatrix}1&2\end{pmatrix} \\
      \bar R = \begin{pmatrix}0.5&1\\2.5&5\end{pmatrix}\\
      \bar Z_1 =\begin{pmatrix}0.5&1\\2.5&5\end{pmatrix}\odot \mathbf{1}[Z_1>0]\\
      \bar Z_1 = \begin{pmatrix}0.5&1\\2.5&5\end{pmatrix}\odot\begin{pmatrix}1&0\\1&1\end{pmatrix}\\
      \bar Z_1 = \begin{pmatrix}0.5&0\\2.5&5\end{pmatrix}\\
      \bar W_1 = \begin{pmatrix}1&2\\3&1\end{pmatrix}^{\!\top} \begin{pmatrix}0.5&0\\2.5&5\end{pmatrix}\\
      \bar W_1 = \begin{pmatrix}1&3\\2&1\end{pmatrix} \begin{pmatrix}0.5&0\\2.5&5\end{pmatrix}\\
      \bar W_1 = \begin{pmatrix}8&15\\3.5&5\end{pmatrix}\\
      \bar{\mathbf{b}_1}= \begin{pmatrix}0.5&0\\2.5&5\end{pmatrix}^{\!\top} \begin{pmatrix}1\\1\end{pmatrix}\\
      \bar{\mathbf{b}_1}= \begin{pmatrix}0.5&2.5\\0&5\end{pmatrix} \begin{pmatrix}1\\1\end{pmatrix}\\
      \bar{\mathbf{b}_1}= \begin{pmatrix}3\\5\end{pmatrix}\\
    \end{gathered}
  \]
  (c) Now let's verify one entry of $\bar W_1$ numerically by evaluating
  \[
    \frac{L(W_1+hE_{kj})-L(W_1-hE_{kj})}{2h}
  \]
  Let's start by calculating the updated $Z_2$ for each variation in $W_1$, setting $k=j=2$:
  \[
    \begin{gathered}
      W_1+hE_{kj}=\begin{pmatrix}0&1\\1&-1+h\end{pmatrix}\\
      Z_{1+h}=X(W_1+hE_{kj})+\mathbf{1}_2\mathbf{b}_1^{\!\top}\\
      Z_{1+h}=XW_1+XhE_{kj}+\mathbf{1}_2\mathbf{b}_1^{\!\top}\\
      Z_{1+h}=XhE_{kj}+Z_1\\
      Z_{1+h}=\begin{pmatrix}1&2\\3&1\end{pmatrix}\begin{pmatrix}0&0\\0&h\end{pmatrix}+Z_1\\
      Z_{1+h}=\begin{pmatrix}0&2h\\0&h\end{pmatrix}+Z_1\\
      Z_{1+h}=\begin{pmatrix}0&2h\\0&h\end{pmatrix}+\begin{pmatrix}3&-1\\2&2\end{pmatrix}\\
      Z_{1+h}=\begin{pmatrix}3&-1+2h\\2&2+h\end{pmatrix}\\
      R_{+h}=\operatorname{relu}(Z_{1+h})\\
      R_{+h}=\begin{pmatrix}3&0\\2&2+h\end{pmatrix}\\
      Z_{2+h}=R_{+h}W_2+\mathbf{1}_2b_2\\
      Z_{2+h}=(R+hE_{kj})W_2+\mathbf{1}_2b_2\\
      Z_{2+h}=RW_2+hE_{kj}W_2+\mathbf{1}_2b_2\\
      Z_{2+h}=hE_{kj}W_2+Z_2\\
      Z_{2+h}=\begin{pmatrix}0&0\\0&h\end{pmatrix}\begin{pmatrix}1\\2\end{pmatrix}+Z_2\\
      Z_{2+h}=\begin{pmatrix}0\\2h\end{pmatrix}+Z_2\\
      Z_{2+h}=\begin{pmatrix}0\\2h\end{pmatrix}+\begin{pmatrix}3\\6\end{pmatrix}\\
      Z_{2+h}=\begin{pmatrix}3\\6+2h\end{pmatrix}\\
      W_1-E_{kj}=\begin{pmatrix}0&1\\1&-1-h\end{pmatrix}\\
      Z_{1-h}=-XhE_{kj}+Z_1\\
      Z_{1-h}=\begin{pmatrix}3&-1-2h\\2&2-h\end{pmatrix}\\
      R_{-h}=\operatorname{relu}(Z_{1-h})=\begin{pmatrix}3&0\\2&2-h\end{pmatrix}\\
      R_{-h}=R-hE_{kj}\\
      Z_{2-h}=(R-hE_{kj})W_2+\mathbf{1}_2b_2\\
      Z_{2-h}=RW_2-hE_{kj}W_2+\mathbf{1}_2b_2\\
      Z_{2-h}=-hE_{kj}W_2+Z_2\\
      Z_{2-h}=\begin{pmatrix}0&0\\0&-h\end{pmatrix}\begin{pmatrix}1\\2\end{pmatrix}+Z_2\\
      Z_{2-h}=\begin{pmatrix}0\\-2h\end{pmatrix}+Z_2\\
      Z_{2-h}=\begin{pmatrix}3\\6-2h\end{pmatrix}\\
    \end{gathered}
  \]
  Now let's evaluate the finite difference:
  \[
    \begin{gathered}
      \frac{\frac{1}{2m}(\norm{Z_{2+h}-Y}^2_F-\norm{Z_{2-h}-Y}^2_F)}{2h}\\
      \frac{\frac{1}{2m}(((2-3)^2+(1-6+2h)^2)-((2-3)^2+(1-6-2h)^2))}{2h}\\
      \frac{\frac{1}{2m}((1+(1-(6+2h))^2)-(1+(1-(6-2h))^2))}{2h}\\
      \frac{\frac{1}{2m}((1-(6+2h))^2-(1-(6-2h))^2)}{2h}\\
      \frac{\frac{1}{4}((1-(6+2h))^2-(1-(6-2h))^2)}{2h}\\
      \frac{((1-6-2h)^2-(1-6+2h)^2)}{8h}\\
      \frac{(-5-2h)^2-(-5+2h)^2)}{8h}\\
      \frac{(25+20h+4h^2)-(25-20h+4h^2)}{8h}\\
      \frac{25+20h+4h^2-25+20h-4h^2}{8h}\\
      \frac{40h}{8h}\\
      5\\
    \end{gathered} 
  \]
  This matches the $(2,2)$ entry of $\bar W_2$.
  (d) Because $Z_1$ has a negative entry, and that becomes a zero entry in $\bar Z_1$. This is because nudging the negative entry does nothing to $R$, since $\operatorname{relu}$ will zero it out anyways. The same does not apply to $R$, since it has already had $\operatorname{relu}$ applied, so the same phenomenon doesn't take place.

\end{solution}
\begin{solution}{ex:ml-3}
  We can compute $\frac{\partial L}{\partial Z_{ij}}$ directly from the sum using the limit definition of the derivative. All other terms except the $Z_{ij}$ term cancel out from the sum, so we get:
  \[
    \begin{gathered}
      \frac{\partial L}{\partial Z_{ij}}=\lim_{h\to 0}\frac{\frac{1}{2m}(((Z_{ij}+h)-Y)^2-(Z_{ij}-Y)^2)}{h}\\
      \frac{\partial L}{\partial Z_{ij}}=\lim_{h\to 0}\frac{(Z_{ij}+h)^2-2Y(Z_{ij}+h)+Y^2- (Z_{ij}^2-2Z_{ij}Y+Y^2)}{2mh}\\
      \frac{\partial L}{\partial Z_{ij}}=\lim_{h\to 0}\frac{(Z_{ij}+h)^2-2Y(Z_{ij}+h)+Y^2- Z_{ij}^2+2Z_{ij}Y-Y^2}{2mh}\\
      \frac{\partial L}{\partial Z_{ij}}=\lim_{h\to 0}\frac{(Z_{ij}+h)^2-2Y(Z_{ij}+h)- Z_{ij}^2+2Z_{ij}Y}{2mh}\\
      \frac{\partial L}{\partial Z_{ij}}=\lim_{h\to 0}\frac{(Z_{ij}^2+2hZ_{ij}+h^2)-2Y(Z_{ij}+h)- Z_{ij}^2+2Z_{ij}Y}{2mh}\\
      \frac{\partial L}{\partial Z_{ij}}=\lim_{h\to 0}\frac{2hZ_{ij}+h^2-2Y(Z_{ij}+h)+2Z_{ij}Y}{2mh}\\
      \frac{\partial L}{\partial Z_{ij}}=\lim_{h\to 0}\frac{2hZ_{ij}+h^2-2YZ_{ij}-2Yh+2Z_{ij}Y}{2mh}\\
      \frac{\partial L}{\partial Z_{ij}}=\lim_{h\to 0}\frac{2hZ_{ij}+h^2-2Yh}{2mh}\\
      \frac{\partial L}{\partial Z_{ij}}=\lim_{h\to 0}\frac{2Z_{ij}+h-2Y}{2m}\\
      \frac{\partial L}{\partial Z_{ij}}=\frac{Z_{ij}-Y}{m}\\
    \end{gathered}
  \]
  The partial derivative could've also been trivially calculated using the chain rule (inner function $g(Z_{ij})=Z_{ij}-Y$ and outer function $f(x)=x^2$), but I calculated it directly from the sum.
\end{solution}
